```ts
/**
 * Le texte des recettes, tel qu'il arrive des sites (R5).
 *
 * Deux défauts relevés dans le catalogue : des entités HTML restées dans le texte
 * (« Fish &amp; chips »), et des titres qui finissent en réclame (« Poulet
 * basquaise : la meilleure recette »). Le premier se lit comme une faute ; le
 * second allonge chaque ligne de la liste pour ne rien dire.
 *
 * Pur et sans réseau : l'ingestion, le recalcul et l'écran s'en servent tous les
 * trois, et ils doivent dire la même chose.
 */

const NOMMEES: Record<string, string> = {
  amp: '&', quot: '"', apos: '\'', lt: '<', gt: '>', nbsp: ' ',
  rsquo: '’', lsquo: '‘', laquo: '«', raquo: '»', hellip: '…', ndash: '–', mdash: '—',
  eacute: 'é', egrave: 'è', ecirc: 'ê', euml: 'ë', agrave: 'à', acirc: 'â',
  ccedil: 'ç', icirc: 'î', iuml: 'ï', ocirc: 'ô', ucirc: 'û', ugrave: 'ù',
  oelig: 'œ', deg: '°',
}

/**
 * « &amp; », « &#039; », « &eacute; » : ce que le JSON-LD a gardé du HTML.
 *
 * Deux passes, parce que l'échappement double existe (« &amp;amp; »). Une entité
 * inconnue reste telle quelle : mieux vaut un « &zz; » visible qu'un caractère
 * inventé.
 */
export function decodeEntites(s: string): string {
  let t = s
  for (let passe = 0; passe < 2; passe++) {
    t = t.replace(/&(#x[0-9a-f]+|#\d+|[a-z]+);/gi, (tout, code: string) => {
      if (code[0] === '#') {
        const n = code[1] === 'x' || code[1] === 'X'
          ? parseInt(code.slice(2), 16)
          : Number(code.slice(1))
        return n > 0 && n < 0x110000 ? String.fromCodePoint(n) : tout
      }
      return NOMMEES[code.toLowerCase()] ?? tout
    })
  }
  return t
}

/** Minuscules, sans accents : on compare des mots, pas des orthographes. */
export function plie(s: string): string {
  return s.toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '')
    .replace(/œ/g, 'oe').replace(/æ/g, 'ae')
}

/** Ce qui sépare le nom du plat de ce que le site y a collé. Le tiret doit être
    entouré d'espaces : « Saint-Jacques » n'est pas un séparateur. */
const SEPARATEUR = String.raw`\s*(?:[:|–—,]|\s-)\s*`
/** Une marque SEULE après le séparateur : un appareil ou un site, pas un plat. */
const MARQUE = String.raw`(?:cookeo|thermomix|companion|monsieur cuisine|air ?fryer|marmiton|ptitchef|cuisineaz|jow)`

const FIN_RECETTE = new RegExp(`${SEPARATEUR}[^:|–—,]*\\brecettes?\\b[^:|–—,]*$`, 'i')
const PARENTHESE_RECETTE = /\s*\([^()]*\brecettes?\b[^()]*\)\s*$/i
const FIN_MARQUE = new RegExp(`${SEPARATEUR}${MARQUE}\\s*$`, 'i')
const DEBUT_RECETTE = /^recettes?\s*(?::\s*|de\s+(?:la\s+|l['’]\s*)?|du\s+|des\s+|d['’]\s*)/i

/**
 * Le titre, sans ce que le site y a ajouté pour les moteurs de recherche.
 *
 * Dans l'ordre : les entités ; une fin introduite par un séparateur et qui
 * contient « recette » (« : la meilleure recette », « - Recette Ptitchef ») ; une
 * parenthèse finale qui en contient ; une marque seule en fin (« - Cookeo ») ;
 * « Recette de… » en tête ; l'accent que le clavier a perdu (« a l' » → « à l' ») ;
 * la majuscule.
 *
 * Prudente : un titre qui ne ressemble à aucun motif passe tel quel, et un
 * nettoyage qui ne laisserait pas trois caractères rend le titre d'origine.
 */
export function titrePropre(brut: string | null): string | null {
  if (brut === null) return null
  const decode = decodeEntites(brut).replace(/\s+/g, ' ').trim()
  const t = decode
    .replace(FIN_RECETTE, '')
    .replace(PARENTHESE_RECETTE, '')
    .replace(FIN_MARQUE, '')
    .replace(DEBUT_RECETTE, '')
    .replace(/(\S) a (l['’]|la )/g, '$1 à $2')
    .replace(/[\s.]+$/, '')
    .trim()
  if (t.length < 3) return decode
  return t[0].toLocaleUpperCase('fr') + t.slice(1)
}
```

```ts
/**
 * Les nœuds du bloc qui portent un `@id`, pour suivre les renvois.
 *
 * ⚠️ Un objet qui ne porte QUE son `@id` est un renvoi, pas un nœud. Dans un graphe
 *    Yoast, l'`Article` et la `WebPage` renvoient à `…#primaryimage` avant que
 *    l'`ImageObject` complet n'arrive : ranger le premier vu gardait le renvoi
 *    vide, et la photo devenait introuvable.
 */
function noeuds(o: unknown, sortie = new Map<string, Record<string, unknown>>()) {
  if (Array.isArray(o)) {
    for (const v of o) noeuds(v, sortie)
  } else if (o && typeof o === 'object') {
    const r = o as Record<string, unknown>
    const id = r['@id']
    const plein = Object.keys(r).some(k => k !== '@id')
    if (typeof id === 'string' && plein && !sortie.has(id)) sortie.set(id, r)
    for (const v of Object.values(r)) noeuds(v, sortie)
  }
  return sortie
}

const RESSEMBLE_A_UNE_IMAGE = /\.(?:jpe?g|png|webp|avif|gif)(?:[?#]|$)/i

/**
 * La photo de la recette, en LIEN : rien n'est copié chez nous (R3).
 *
 * `image` s'écrit de quatre façons — une adresse, une liste d'adresses, un
 * `ImageObject` (`url`, puis `contentUrl`), une liste d'`ImageObject`. On prend
 * la PREMIÈRE adresse utilisable : celle que le site met en avant.
 *
 * Un objet qui ne porte qu'un `@id` renvoie ailleurs dans le graphe : on l'y
 * cherche. Introuvable, on ne le garde que s'il finit comme une image — sinon
 * c'est l'ancre d'une page (`…/#primaryimage`), pas une photo.
 *
 * L'adresse est décodée, résolue contre la page, passée en https (le navigateur
 * le ferait de toute façon, l'appli étant servie en https), et refusée au-delà
 * de 2 048 caractères — la limite de la colonne.
 */
export function photo(
  v: unknown, page: string, graphe = new Map<string, Record<string, unknown>>(),
): string | null {
  const candidates: string[] = []
  const visite = (x: unknown, profondeur: number): void => {
    if (profondeur > 3) return
    if (typeof x === 'string') { candidates.push(x); return }
    if (Array.isArray(x)) { for (const e of x) visite(e, profondeur + 1); return }
    if (!x || typeof x !== 'object') return
    const o = x as Record<string, unknown>
    const directe = typeof o.url === 'string' ? o.url
      : typeof o.contentUrl === 'string' ? o.contentUrl : null
    if (directe) { candidates.push(directe); return }
    const id = typeof o['@id'] === 'string' ? o['@id'] : null
    if (!id) return
    const renvoi = graphe.get(id)
    if (renvoi && renvoi !== o) { visite(renvoi, profondeur + 1); return }
    if (RESSEMBLE_A_UNE_IMAGE.test(id)) candidates.push(id)
  }
  visite(v, 0)

  for (const c of candidates) {
    const brut = decodeEntites(c.trim())
    if (!brut) continue
    let u: URL
    try { u = new URL(brut, page) } catch { continue }
    if (u.protocol === 'http:') u.protocol = 'https:'
    if (u.protocol !== 'https:') continue
    const adresse = u.toString()
    if (adresse.length <= 2048) return adresse
  }
  return null
}
```

```ts
/**
 * Les formes d'un geste DEMANDÉ, en trois familles selon où elles valent (R6).
 *
 *  · `partout` — l'infinitif et l'impératif en « vous » : « verser », « versez ».
 *    Ils ne décrivent jamais un ingrédient.
 *  · `enTete` — l'impératif en « tu » (Jow) : « verse », « garnis ». C'est aussi
 *    la forme d'un NOM : « le beurre », « du mélange », « de poivre ». On ne la
 *    croit qu'en tête de proposition, ou après « et », « puis »…
 *  · `apresEn` — le gérondif : « en remuant ». Sans « en » devant, c'est un
 *    adjectif : « un couteau bien tranchant ».
 *
 * Le participe passé n'est dans aucune : « les poivrons coupés » décrit des
 * poivrons, « l'eau salée » de l'eau.
 */
function formesDeGeste(infinitif: string) {
  if (infinitif.endsWith('er')) return { partout: 'er|ez', enTete: 'e', apresEn: 'ant|eant' }
  // couvrir : couvrir, couvrez, couvre, couvrant
  if (/[vf]rir$/.test(infinitif)) return { partout: 'ir|ez', enTete: 'e', apresEn: 'ant' }
  // garnir (radical « garni ») : garnir, garnissez, garnis, garnissant
  if (infinitif.endsWith('ir')) return { partout: 'r|ssez', enTete: 's', apresEn: 'ssant' }
  // battre (radical « batt ») : battre, battez, battant
  return { partout: 're|ez|sez', enTete: 's', apresEn: 'ant|sant' }
}

/** Les mots après lesquels un impératif en « tu » ouvre encore une proposition. */
const LIANTS = new Set(['et', 'puis', 'ensuite', 'enfin', 'alors', 'ou'])

type Formes = { partout: RegExp; enTete: RegExp; apresEn: RegExp }
type Geste = { canonique: string; ligne: LigneDuree; formes: Formes | null; phrase: RegExp | null }

/**
 * Les verbes de MAIN du référentiel : ceux qui n'occupent aucun appareil et ne
 * sont pas passifs. « mixer » n'en est pas un — le blender travaille seul —,
 * mais « verser » le lait dans le blender, si.
 */
function indexGestes(ref: Referentiel): Geste[] {
  const mains = new Map<string, LigneDuree>()
  for (const d of ref.durees) {
    if (d.appliance_type === null && d.load_type !== 'passif' && !mains.has(d.verb)) {
      mains.set(d.verb, d)
    }
  }
  const verbes = new Set<string>([...ref.durees.map(d => d.verb), ...Object.keys(ref.alias)])
  const gestes: Geste[] = []
  for (const verbe of verbes) {
    const canonique = ref.alias[verbe] ?? verbe
    const ligne = mains.get(canonique)
    if (!ligne) continue
    const infinitif = pliure(verbe)
    if (infinitif.includes(' ')) {
      // « séparer les blancs », « retirer du feu » : on cherche la phrase entière.
      gestes.push({ canonique, ligne, formes: null, phrase: motifDe(verbe) })
      continue
    }
    const tete = radical(verbe)
    const f = formesDeGeste(infinitif)
    gestes.push({
      canonique, ligne, phrase: null,
      formes: {
        partout: new RegExp(`^${tete}(?:${f.partout})$`),
        enTete: new RegExp(`^${tete}(?:${f.enTete})$`),
        apresEn: new RegExp(`^${tete}(?:${f.apresEn})$`),
      },
    })
  }
  return gestes
}

/** Les pronoms qu'on enjambe pour lire ce qui précède un verbe : « sans LES éplucher ». */
const PRONOMS = new Set(['le', 'la', 'les', 'l', 'en', 'y', 'lui', 'leur', 'se', 's'])

/**
 * Le verbe est-il CONJUGUÉ dans le texte, sous une forme qui demande le geste ?
 *
 * Un infinitif nié ou remis à plus tard ne le demande pas ici : « sans les
 * éplucher », « laissez refroidir avant de démouler » — le démoulage a son étape.
 */
function conjugue(texte: string, f: Formes): boolean {
  for (const proposition of texte.split(/[.,;:!?()]+/)) {
    const mots = proposition.split(/[^\p{L}]+/u).filter(Boolean)
    for (let i = 0; i < mots.length; i++) {
      const mot = pliure(mots[i])
      const avant = i > 0 ? pliure(mots[i - 1]) : null
      let k = i - 1
      while (k >= 0 && PRONOMS.has(pliure(mots[k]))) k--
      const precedent = k >= 0 ? pliure(mots[k]) : null
      const ecarte = precedent === 'sans'
        || ((precedent === 'de' || precedent === 'd') && k >= 1 && pliure(mots[k - 1]) === 'avant')
      if (f.partout.test(mot) && !ecarte) return true
      // « émincé » se plie en « emince », comme l'impératif « émince » : seul
      // l'accent du texte d'origine les distingue.
      if (f.enTete.test(mot) && !/é$/iu.test(mots[i])
          && (avant === null || LIANTS.has(avant))) return true
      if (f.apresEn.test(mot) && avant === 'en') return true
    }
  }
  return false
}

let cacheRefGestes: Referentiel | null = null
let cacheGestes: Geste[] = []

/**
 * Les minutes de MAIN qu'une étape demande, verbe par verbe (R6).
 *
 * Le verbe retenu ne suffit pas à le dire : « Dans un mixeur, versez le lait, la
 * banane et les flocons. Mixez. » retient « mixer », parce que le verbe qui
 * mobilise un appareil gagne — et « mixer » est passif. Toutes les étapes d'un
 * smoothie étaient ainsi : zéro minute de travail, et la recette en tête du tri.
 *
 * On additionne donc la durée de référence de chaque verbe de main conjugué dans
 * l'étape, une fois chacun, mise à l'échelle comme l'étape. `null` quand il n'y
 * en a aucun : « Préchauffez le four » ne demande rien.
 */
export function gestesMin(brut: string, ref: Referentiel): number | null {
  if (cacheRefGestes !== ref) { cacheRefGestes = ref; cacheGestes = indexGestes(ref) }
  const texte = brut.normalize('NFC').trim()
  if (!texte) return null
  const plie = pliure(texte)
  const quantite = quantiteEcrite(texte)
  const minutes = new Map<string, number>()
  for (const g of cacheGestes) {
    if (minutes.has(g.canonique)) continue
    const demande = g.phrase ? g.phrase.test(plie) : conjugue(texte, g.formes!)
    if (!demande) continue
    const facteur = g.ligne.scaling === 'lineaire_plafonne' && quantite
      ? Math.min(3, Math.max(1, quantite / 250))
      : 1
    minutes.set(g.canonique, Number(g.ligne.base_minutes) * facteur)
  }
  if (minutes.size === 0) return null
  return Math.round([...minutes.values()].reduce((s, m) => s + m, 0) * 10) / 10
}
```

```ts
/**
 * Une part qui ne peut pas exister (R7).
 *
 * Les bornes visent les ERREURS — un nombre de parts mal lu, un poids faux —, pas
 * les petites parts : une truffe protéinée de 20 g est une vraie part.
 *
 *  · plus de 900 kcal pour 100 g : plus gras que l'huile ;

 *  · moins de 20 ou plus de 2 000 kcal : un verre d'eau, ou une journée entière.
 *
 * Hors de ces bornes, on ne garde PAS de macros : une valeur fausse sert à
 * filtrer, et elle est pire qu'aucune (D18).
 */
export const PART_PLAUSIBLE = {
  kcalMin: 20, kcalMax: 2000, grammesMax: 1500, densiteMax: 900,
} as const

export function plausible(n: { grams: number; kcal: number }): boolean {
  const g = Number(n.grams)
  const k = Number(n.kcal)
  if (!(g > 0) || !Number.isFinite(k)) return false
  if (k < PART_PLAUSIBLE.kcalMin || k > PART_PLAUSIBLE.kcalMax) return false
  if (g > PART_PLAUSIBLE.grammesMax) return false
  return (k / g) * 100 <= PART_PLAUSIBLE.densiteMax
}
```

```ts
/**
 * Le temps des MAINS, d'une étape et d'une recette (R6, R7).
 *
 * La même règle que `tg_recipe_agrege` (migration 0092), écrite une seconde fois
 * parce que la fiche et le recalcul à sec en ont besoin sans base. Le banc de la
 * base (`tests/recettes.test.ts`) vérifie que les deux disent la même chose : si
 * l'une change, l'autre doit suivre.
 */
export type Charge = 'actif' | 'passif' | 'bloquant'

/** Ce qu'il faut d'une étape pour compter ses minutes. PostgREST rend parfois les
    `numeric` en chaîne : on accepte les deux. */
export type EtapeTemps = {
  load_type: string | null
  duration_min: number | string | null
  gesture_min: number | string | null
}

const nombre = (v: number | string | null | undefined): number =>
  v === null || v === undefined || v === '' ? 0 : Number(v)

/**
 * Active ou bloquante : sa durée, ou ses gestes s'ils la dépassent.
 * Passive : ses gestes seulement — le four cuit seul, mais quelqu'un a versé la
 * pâte dans le moule.
 */
export function minutesDeMains(e: EtapeTemps): number {
  if (e.load_type === 'actif' || e.load_type === 'bloquant') {
    return Math.max(nombre(e.duration_min), nombre(e.gesture_min))
  }
  return nombre(e.gesture_min)
}

export function tempsActif(etapes: EtapeTemps[]): number {
  return Math.round(etapes.reduce((s, e) => s + minutesDeMains(e), 0) * 10) / 10
}

/**
 * Un temps actif nul ne se croit pas : aucune recette ne se fait sans les mains.
 * Il veut dire « aucun geste reconnu », et se lit « inconnu » (R6).
 */
export function actifConnu(min: number | string | null | undefined): boolean {
  return min !== null && min !== undefined && nombre(min) > 0
}

/**
 * « En tout », dans la fiche (R7).
 *
 * Le total vient du site, le temps actif de nos étapes. Quand le site annonce
 * moins que nos gestes, l'un des deux a tort — on ne finit pas avant d'avoir fini
 * ses gestes. On montre alors la somme de nos étapes, chacune comptée au moins
 * pour ses gestes : c'est déjà ce qu'une recette collée appelle « en tout ». La
 * donnée, elle, ne change pas.
 */
export function enTout(
  declare: number | string | null, actif: number | string | null, etapes: EtapeTemps[],
): number | null {
  const site = declare === null ? null : nombre(declare)
  if (site !== null && site > 0 && site >= nombre(actif)) return site
  const somme = etapes.reduce((s, e) => s + Math.max(nombre(e.duration_min), minutesDeMains(e)), 0)
  return somme > 0 ? Math.round(somme * 10) / 10 : null
}

/** Ce que la pilule d'une étape dit de sa charge, dans la fiche. */
export const MOT_DE_CHARGE: Record<Charge, string> = {
  actif: 'tes gestes',
  bloquant: 'à surveiller',
  passif: 'sans toi',
}
```

```sql
-- ═══════════════════════════════════════════════════════════════════════════
-- LES RECETTES, REFAITES (spec du 19/10/2026 : R3, R6, et « avec ce que j'ai »)
--
-- Trois choses, toutes au service du nouvel écran « Choisir » :
--
--  1. la PHOTO de chaque recette, en lien vers celle du site — jamais copiée ;
--  2. les GESTES d'une étape : « Dans un mixeur, versez le lait… Mixez »
--     comptait zéro minute de travail, parce que le verbe qui mobilise un
--     appareil l'emporte et qu'il est passif. 73 recettes affichaient
--     « 0 min actives » et passaient en tête du tri ;
--  3. « Avec ce que j'ai » cherché sur TOUT le catalogue, et non plus parmi les
--     quatre-vingts recettes les plus rapides.
-- ═══════════════════════════════════════════════════════════════════════════

-- ── 1 · La photo ────────────────────────────────────────────────────────────
alter table public.recipe
  add column image_url text
    check (image_url is null
           or (image_url ~ '^https://' and length(image_url) <= 2048));

comment on column public.recipe.image_url is
  'La photo du site d''origine, en LIEN : rien n''est copié chez nous. Lue dans '
  'le JSON-LD (champ image). Écrite par le rôle de service seul (tg_recipe_photo).';

/**
 * La photo ne se saisit pas.
 *
 * Chaque foyer charge la photo de chaque recette qu'il voit. Une adresse posée
 * par un compte — sur le catalogue, que tous peuvent corriger (classe B), ou
 * sur une recette « publique » — ferait charger à tous une image de son choix,
 * et lui dirait qui regarde et quand. Seule l'ingestion l'écrit, depuis la page
 * de la recette.
 *
 * `is_service_role()` et pas `current_user` : la fonction n'est pas `security
 * definer`, mais elle ne doit pas se taire le jour où elle le deviendrait.
 */
create or replace function public.tg_recipe_photo()
returns trigger language plpgsql as $$
begin
  if public.is_service_role() then return new; end if;
  if tg_op = 'INSERT' then
    if new.image_url is not null then
      raise exception 'la photo d''une recette vient de sa page, pas d''une saisie'
        using errcode = 'check_violation';
    end if;
  elsif new.image_url is distinct from old.image_url then
    raise exception 'la photo d''une recette vient de sa page, pas d''une saisie'
      using errcode = 'check_violation';
  end if;
  return new;
end $$;

create trigger recipe_photo before insert or update of image_url on public.recipe
  for each row execute function public.tg_recipe_photo();

-- ── 2 · Les gestes d'une étape ──────────────────────────────────────────────
alter table public.recipe_step
  add column gesture_min numeric check (gesture_min >= 0);

comment on column public.recipe_step.gesture_min is
  'Minutes de gestes que l''étape conjugue (« versez », « mélangez »), quel que '
  'soit le verbe retenu. NULL = non mesuré : l''étape compte comme avant 0092.';

/**
 * Le temps actif compte AU MOINS les gestes de chaque étape (R6).
 *
 * Une étape active ou bloquante compte sa durée, ou ses gestes s'ils la
 * dépassent. Une étape passive ne compte que ses gestes : le four cuit seul,
 * mais quelqu'un a versé la pâte dans le moule.
 *
 * `gesture_min` nul laisse le calcul d'avant intact : les recettes collées et
 * inventées, et tout ce qui n'a pas encore été recalculé.
 *
 * La même règle vit dans `src/lib/recettes/temps.ts` (`minutesDeMains`) ; un
 * banc vérifie que les deux disent la même chose.
 */
create or replace function public.tg_recipe_agrege()
returns trigger language plpgsql security definer set search_path = public as $$
declare r uuid := coalesce(new.recipe_id, old.recipe_id);
begin
  update public.recipe set
    active_time_min = (
      select coalesce(sum(case
               when load_type in ('actif', 'bloquant')
                 then greatest(coalesce(duration_min, 0), coalesce(gesture_min, 0))
               else coalesce(gesture_min, 0)
             end), 0)
      from public.recipe_step where recipe_id = r),
    appliances = (
      select coalesce(array_agg(distinct appliance_type), '{}')
      from public.recipe_step
      where recipe_id = r and appliance_type is not null),
    step_count = (select count(*) from public.recipe_step where recipe_id = r)
  where id = r;
  return null;
end $$;
```

```ts
/**
 * Ce qu'une page relue nous dit, et quand laisser un site tranquille (R3).
 *
 * La relecture des pages déjà ingérées ne cherche qu'une chose : la photo. Elle
 * ne force rien. Un site qui refuse — par un statut, ou par une page de défi
 * sans recette, qui répond souvent 200 — est laissé après trois refus DE SUITE.
 * Une page disparue n'est pas un refus : le site répond, la recette n'y est plus.
 */
export type Issue = 'photo' | 'sans-photo' | 'sans-recette' | 'disparue' | 'refusee' | 'erreur'

export const REFUS_AVANT_ABANDON = 3

export function issueDe(statut: number, aUneRecette: boolean, photo: string | null): Issue {
  if (statut === 404 || statut === 410) return 'disparue'
  if (statut === 401 || statut === 403 || statut === 429 || statut === 503) return 'refusee'
  // 0 : le réseau, ou le délai de 30 s.
  if (statut < 200 || statut >= 300) return 'erreur'
  if (!aUneRecette) return 'sans-recette'
  return photo ? 'photo' : 'sans-photo'
}

const estUnRefus = (i: Issue) => i === 'refusee' || i === 'sans-recette'

/** Trois refus de suite, et on laisse le site pour cette fois. */
export function abandonne(dernieres: Issue[]): boolean {
  const fin = dernieres.slice(-REFUS_AVANT_ABANDON)
  return fin.length === REFUS_AVANT_ABANDON && fin.every(estUnRefus)
}
```

```js
  for (let de = 0; ; de += 1000) {
    const { data, error } = await db.from('recipe')
      .select('id, source_url')
      .is('image_url', null).is('owner_household_id', null)
      .eq('origin', 'importee').not('source_url', 'is', null)
      .order('id').range(de, de + 999)

}

if (DOMAINES.length > 0) recettes = recettes.filter(r => DOMAINES.includes(hote(r.source_url)))

 * dans l'ordre des `id`, trois pages mortes en tête d'un site le feraient

  }

  const robots = lireRobots(await texteDe(`${new URL(liste[0].source_url).origin}/robots.txt`, 15000))
  await dors(PAUSE_MS)
  // Un `Crawl-delay` plus long que le nôtre l'emporte.

  const issues = []

  for (const [i, r] of liste.entries()) {
    if (abandonne(issues)) { b.abandonnees = liste.length - i; break }
    if (!autorise(new URL(r.source_url).pathname, robots)) { b.interdites++; continue }

    let statut = 0
    let html = null
    try {
      const res = await fetch(r.source_url, {
        headers: { 'User-Agent': AGENT, 'Accept-Language': 'fr-FR,fr;q=0.9' },
        signal: AbortSignal.timeout(30000),
        redirect: 'follow',
      })
      statut = res.status
      if (res.ok) html = await res.text()
    } catch {
      statut = 0
    }
    const brute = html ? extraire(html, r.source_url) : null
    const issue = issueDe(statut, brute !== null, brute?.image ?? null)

  }
  bilan.push(b)
}
```

```js
/** PostgREST plafonne à 1 000 lignes en silence. On pagine, toujours. */
async function tout(table, select, filtre = q => q, cle = 'id') {
  const pages = []
  for (let de = 0; ; de += 1000) {
    // ⚠️ `order('id')` n'est pas décoratif : sans ordre TOTAL, deux pages
    //    successives peuvent sauter des lignes ou en rendre deux fois. Sur
    //    17 000 ingrédients, la couverture se calculait alors sur une liste
    //    tronquée et sortait à 1.0 sur un quart de plat.
    const { data, error } = await filtre(db.from(table).select(select))
      .order(cle).range(de, de + 999)
    if (error) throw new Error(`${table} : ${error.message}`)
    pages.push(...data)
    if (data.length < 1000) break
  }
  return pages
}

/** Écrit par paquets de deux cents : assez pour aller vite, pas assez pour tout bloquer. */
async function parPaquets(lignes, ecrire, quoi) {
  for (let d = 0; d < lignes.length; d += 200) {
    await Promise.all(lignes.slice(d, d + 200).map(ecrire))
    process.stdout.write(`   ${Math.min(d + 200, lignes.length)}/${lignes.length}\r`)
  }
  console.log(`   ${lignes.length}/${lignes.length} ${quoi}.      `)
}

const mediane = l => {
  if (l.length === 0) return 0
  const t = [...l].sort((a, b) => a - b)
  return Math.round(t[Math.floor(t.length / 2)] * 10) / 10
}

  }
```

```ts
 * On ne fait que descendre : « Les 10 suivantes » est le geste qu'on fait neuf
 * fois sur dix. Les numéros servent à revenir — le premier, le dernier, la page
 * et ses voisines, des « … » entre. Sept boutons au plus : ils tiennent sur un
 * téléphone, à 44 px chacun.
 */
export const PAR_PAGE = 10

export const nombreDePages = (total: number) => Math.max(1, Math.ceil(total / PAR_PAGE))

/** Les recettes de la page. Une page au-delà de la fin rend la dernière. */
export function tranche<T>(liste: T[], page: number): T[] {
  const p = Math.min(Math.max(1, page), nombreDePages(liste.length))
  return liste.slice((p - 1) * PAR_PAGE, p * PAR_PAGE)
}

/** « Recettes 11 à 20 sur 1 390 ». Liste vide : de 0 à 0. */
export function bornesDeLaPage(page: number, total: number): { de: number; a: number } {
  if (total === 0) return { de: 0, a: 0 }
  const p = Math.min(Math.max(1, page), nombreDePages(total))
  return { de: (p - 1) * PAR_PAGE + 1, a: Math.min(p * PAR_PAGE, total) }
}

export function numeros(courante: number, total: number): (number | '…')[] {
  if (total <= 5) return Array.from({ length: total }, (_, i) => i + 1)
  const gardees = new Set([1, total, courante - 1, courante, courante + 1])
  // Aux deux bouts, on montre trois pages d'affilée : « 1 2 3 … 139 ».
  if (courante <= 2) { gardees.add(2); gardees.add(3) }
  if (courante >= total - 1) { gardees.add(total - 1); gardees.add(total - 2) }
  const triees = [...gardees].filter(n => n >= 1 && n <= total).sort((a, b) => a - b)
  const sortie: (number | '…')[] = []
  triees.forEach((n, i) => {
    if (i > 0 && n - triees[i - 1] > 1) sortie.push('…')
    sortie.push(n)
  })
  return sortie
}
```

```ts
 * L'ordre du parcours est une exigence, pas une préférence : **le gratuit

 * que j'ai » et « Jamais essayée » cherchent enfin dans tout le catalogue. Ce qui
 * pèse — les ingrédients, les étapes — ne se charge que pour ce qui est affiché.
 *

 *
 *  · **Le temps ACTIF, pas le temps total.** Dix minutes de gestes et quarante
 *    de four ne font pas cinquante minutes de travail (D19, D30).
 *  · **La borne DÉFAVORABLE, jamais la moyenne** (D18). « Au moins 30 g de

/** Une recette du catalogue, en léger : ce qu'il faut à la liste et aux filtres. */
export type Candidate = {

  title: string | null
  /** La photo du site, en lien (R3). */

  total_time_min: number | null

/** Les huit appareils de l'écran : ceux qu'on coche et qu'on dessine (R4). */
export const CODES_APPAREILS = ['four', 'plaques', 'air_fryer', 'micro_ondes',
  'autocuiseur', 'blender', 'batteur', 'robot_cuiseur'] as const

/** Les bornes affichables d'une recette. Jamais la moyenne (D18). */
export function bornes(c: Candidate) {
  if (!c.nutrition) return null
  return {
    proteinesMin: Math.round(c.nutrition.protein_g - c.nutrition.protein_g_margin),
    proteinesMax: Math.round(c.nutrition.protein_g + c.nutrition.protein_g_margin),
    kcalMin: Math.round(c.nutrition.kcal - c.nutrition.kcal_margin),
    kcalMax: Math.round(c.nutrition.kcal + c.nutrition.kcal_margin),

  }

/** Ce que les filtres consultent hors de la recette elle-même. */
export type Contexte = {
  /** Les recettes que le foyer a déjà cuisinées. */
  dejaFaites: Set<string>
  /** Les recettes faisables avec le placard ; `null` tant qu'on ne sait pas. */
  faisables: Set<string> | null
}

/**
 * Le candidat passe-t-il les filtres ?
 *
 * Extrait pour être éprouvé : c'est ici que vit la règle de la BORNE
 * DÉFAVORABLE, et une inversion de signe y passerait inaperçue — l'écran
 * afficherait simplement quelques recettes de trop.
 */
export function retient(c: Candidate, f: Filtres, ctx: Contexte): boolean {
  const texte = plie(f.texte.trim())
  if (texte && !plie(c.title ?? '').includes(texte)) return false

  // Un temps inconnu ne tient aucune promesse de temps (R6).
  if (f.tempsActifMax !== null
      && !(actifConnu(c.active_time_min) && Number(c.active_time_min) <= f.tempsActifMax)) {
    return false
  }
  // « On ne sait pas » n'est pas « oui » : seul `true` se congèle (D27).
  if (f.congelable === true && c.freezable !== true) return false
  // Un appareil non coché ne doit apparaître dans AUCUNE étape. Un code que
  // l'écran ne dessine pas ne compte pas : la base ne le comptait pas non plus.
  if (f.appareils.length > 0 && c.appliances.some(a =>
    (CODES_APPAREILS as readonly string[]).includes(a) && !f.appareils.includes(a))) {
    return false
  }
  if (f.jamaisFaites && ctx.dejaFaites.has(c.id)) return false
  if (f.avecCeQuOnA && !(ctx.faisables?.has(c.id) ?? false)) return false

  // Une recette sans macros connues n'est pas écartée par défaut : elle ne
  // l'est que si l'on filtre justement sur ce qu'on ignore.
  if (f.proteinesMin === null && f.kcalMax === null) return true
  const b = bornes(c)
  if (!b) return false
  // Protéines : la borne BASSE doit atteindre le seuil.
  if (f.proteinesMin !== null && b.proteinesMin < f.proteinesMin) return false
  // Calories : la borne HAUTE doit rester sous le plafond.
  if (f.kcalMax !== null && b.kcalMax > f.kcalMax) return false
  return true
}

const ORDRE_DES_TITRES = new Intl.Collator('fr')

/**
 * Le temps actif croissant, l'inconnu à la fin, puis le titre, puis l'id.
 *
 * Un ordre TOTAL : sans lui, deux chargements pourraient ranger différemment
 * deux recettes à 12 minutes, et une recette changerait de page sous les yeux.
 */
export function trie(liste: Candidate[]): Candidate[] {
  return [...liste].sort((a, b) => {
    const ka = actifConnu(a.active_time_min)
    const kb = actifConnu(b.active_time_min)
    if (ka !== kb) return ka ? -1 : 1
    if (ka && Number(a.active_time_min) !== Number(b.active_time_min)) {
      return Number(a.active_time_min) - Number(b.active_time_min)
    }
    return ORDRE_DES_TITRES.compare(a.title ?? '', b.title ?? '') || a.id.localeCompare(b.id)
  })
}

/** Combien de filtres sont posés. Sert à la pastille du bouton des filtres. */
export function nombreDeFiltres(f: Filtres): number {
  return [f.tempsActifMax, f.proteinesMin, f.kcalMax, f.congelable].filter(x => x !== null).length
    + (f.appareils.length > 0 ? 1 : 0)
    + (f.avecCeQuOnA ? 1 : 0)
    + (f.jamaisFaites ? 1 : 0)
}

/** Les parts qu'« Ajouter » pose : ce qui manque, borné par le rendement, 2 à défaut. */
export function partsAProposer(cible: number, prises: number, rendement: number | null): number {
  return Math.max(1, Math.min(cible - prises || 2, rendement ?? 2))
}

/**

 * Une ligne non rattachée ne manque pas : on ne sait pas, on se tait.
 */
export function manquesDe(
  lignes: { recipe_id: string; food_id: string | null; raw_text: string }[],
  placard: Set<string>,
): Map<string, string[]> {
  const parRecette = new Map<string, string[]>()
  for (const l of lignes) {
    if (!l.food_id || placard.has(l.food_id)) continue
    const deja = parRecette.get(l.recipe_id) ?? []
    deja.push(l.raw_text)
    parRecette.set(l.recipe_id, deja)
  }
  return parRecette
}

  recipe_nutrition: Candidate['nutrition'] | Candidate['nutrition'][] | null

const PAGE_DE_LA_BASE = 1000

const COLONNES = `id, title, image_url, yield_servings, total_time_min, active_time_min,
  appliances, freezable, owner_household_id, created_by,
  recipe_nutrition(kcal, protein_g, kcal_margin, protein_g_margin, fiber_g, coverage)`

/** Tout le catalogue planifiable, en léger, trié (R1). */
export function useCatalogue() {
  return useQuery({
    queryKey: ['catalogue-leger'],
    staleTime: 10 * 60_000,
    queryFn: async (): Promise<Candidate[]> => {
      const lignes: LigneBrute[] = []
      for (let de = 0; ; de += PAGE_DE_LA_BASE) {
        const page = ou(await supabase.from('recipe').select(COLONNES)
          .not('title', 'is', null)
          .eq('plannable', true)
          .order('id')
          .range(de, de + PAGE_DE_LA_BASE - 1)) as unknown as LigneBrute[]
        lignes.push(...page)
        if (page.length < PAGE_DE_LA_BASE) break
      }

      /*
       * La provenance, quand la recette vient d'un foyer.
       *
       * `user_profile` est cadrée sur le foyer — on n'a pas à lire la liste des
       * membres d'un ami. Mais si l'on voit sa recette, on doit pouvoir en voir
       * l'origine : `prenoms_visibles` ne rend que le prénom, et seulement pour
       * les foyers qu'on a le droit de connaître.
       */
      const prenoms = new Map<string, string>()
      let monFoyer: string | null = null
      if (lignes.some(l => l.owner_household_id !== null)) {
        monFoyer = ou(await supabase.rpc('current_household'))
        for (const p of ou(await supabase.rpc('prenoms_visibles'))) {
          if (p.id && p.display_name) prenoms.set(p.id, p.display_name)
        }
      }

      return trie(lignes.map(({ created_by, recipe_nutrition, ...l }) => {
        // La jointure rend un objet ou un tableau selon la cardinalité déduite.
        const n = Array.isArray(recipe_nutrition) ? recipe_nutrition[0] ?? null : recipe_nutrition
        return {
          ...l,
          appliances: l.appliances ?? [],
          nutrition: n ?? null,
          ajoutePar: created_by ? prenoms.get(created_by) ?? null : null,
          dUnAmi: l.owner_household_id !== null && l.owner_household_id !== monFoyer,
        }
      }))
    },
  })
}
```

```ts
/**
 * La fiche (i) d'une recette, chargée à l'ouverture (R2).
 *
 * Une requête : la source, les ingrédients, les étapes. La liste légère a déjà
 * le reste — photo, temps, parts, macros — et la fiche s'ouvre dessus sans
 * attendre.
 */
import { useQuery } from '@tanstack/react-query'
import { ouNul, supabase } from '../supabase.ts'

export type EtapeDeFiche = {
  ordinal: number
  text: string
  duration_min: number | null
  load_type: 'actif' | 'passif' | 'bloquant' | null
  appliance_type: string | null
  gesture_min: number | null
}

export type Fiche = {
  id: string
  source_name: string | null
  source_url: string | null
  ingredients: { ordinal: number; raw_text: string }[]
  etapes: EtapeDeFiche[]
}

type Brute = Pick<Fiche, 'id' | 'source_name' | 'source_url'> & {
  recipe_ingredient: Fiche['ingredients'] | null
  recipe_step: EtapeDeFiche[] | null
}

const parOrdinal = <T extends { ordinal: number }>(l: T[]) =>
  [...l].sort((a, b) => a.ordinal - b.ordinal)

export function useFiche(id: string | null) {
  return useQuery({
    queryKey: ['fiche', id],
    enabled: id !== null,
    staleTime: 10 * 60_000,
    // Une recette devenue illisible ne le redeviendra pas en réessayant.
    retry: 1,
    queryFn: async (): Promise<Fiche> => {
      const r = ouNul(await supabase.from('recipe')
        .select(`id, source_name, source_url,
                 recipe_ingredient(ordinal, raw_text),
                 recipe_step(ordinal, text, duration_min, load_type, appliance_type, gesture_min)`)
        .eq('id', id!).maybeSingle()) as unknown as Brute | null
      // La recette d'un ami dont le partage vient d'être retiré : elle était dans
      // la liste, elle n'est plus lisible.
      if (!r) throw new Error('Cette recette n’est plus accessible.')
      return {
        id: r.id,
        source_name: r.source_name,
        source_url: r.source_url,
        ingredients: parOrdinal(r.recipe_ingredient ?? []),
        etapes: parOrdinal(r.recipe_step ?? []),
      }
    },
  })
}
```
